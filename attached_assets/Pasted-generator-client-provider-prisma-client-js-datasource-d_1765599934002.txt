generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id           String   @id
  createdAt    DateTime @default(now())
  promptCredits Int     @default(0)
  gumballs     Gumball[]
  jobs         Job[]
  transactions Transaction[]
}

model Product {
  id          String   @id
  name        String
  pulls       Int
  priceRays   Int      // canonical
  active      Boolean  @default(true)
  createdAt   DateTime @default(now())
}

model Transaction {
  id              String   @id
  userId          String
  productId       String
  currency        String   // USD | BTC | SOLAR
  status          String   // PENDING | CONFIRMED | FAILED
  checkoutUrl     String?
  createdAt       DateTime @default(now())
  confirmedAt     DateTime?

  // Snapshot pricing
  priceRays       Int
  pullsPurchased  Int

  user    User    @relation(fields: [userId], references: [id])
  product Product @relation(fields: [productId], references: [id])
}

model Gumball {
  id          String   @id
  userId      String
  title       String
  type        String   // sora_prompt_bundle (etc)
  promptMain  String
  remixJson   String   // JSON string
  mcpRunbook  String   // JSON string
  createdAt   DateTime @default(now())

  user User @relation(fields: [userId], references: [id])
  jobs Job[]
}

model Job {
  id         String   @id
  userId     String
  gumballId  String
  status     String   // QUEUED | RUNNING | AWAITING_ASSET | DELIVERED | FAILED
  provider   String   // SORA_MANUAL for now
  composedPrompt String
  selectedRemixId String
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt

  user    User    @relation(fields: [userId], references: [id])
  gumball Gumball @relation(fields: [gumballId], references: [id])
  assets  Asset[]
}

model Asset {
  id        String   @id
  jobId     String
  kind      String   // video_mp4 | thumbnail_jpg | prompt_txt
  url       String?
  createdAt DateTime @default(now())

  job Job @relation(fields: [jobId], references: [id])
}
