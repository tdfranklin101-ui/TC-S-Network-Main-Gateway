from fastapi import APIRouter, Depends, HTTPException, status

from voucher_service import VoucherService
from voucher_models import VoucherType, VoucherStatus, RedemptionMethod

# Initialize router

router = APIRouter(prefix=”/api/vouchers”, tags=[“vouchers”])

# REQUEST/RESPONSE MODELS

class VoucherTypeEnum(str, Enum):
GOODS_PICKUP = “goods_pickup”
SERVICE = “service”
ADMISSION = “admission”
EXPERIENCE = “experience”
RENTAL = “rental”
SUBSCRIPTION = “subscription”

class CreateVoucherListingRequest(BaseModel):
title: str = Field(…, max_length=200)
description: str
voucher_type: VoucherTypeEnum
price_rays: int = Field(…, gt=0)
redemption_location: str
redemption_instructions: str
redemption_hours: Optional[str] = None
quantity_available: Optional[int] = Field(None, gt=0)
valid_from: Optional[datetime] = None
valid_until: Optional[datetime] = None
redemption_window_hours: Optional[int] = None
category: Optional[str] = None
tags: Optional[List[str]] = None
images: Optional[List[str]] = None
transferable: bool = False
terms_conditions: Optional[str] = None

class ListingProposalResponse(BaseModel):
listing_id: Optional[str] = None
proposal: Dict[str, Any]
requires_approval: bool
message: str

class PurchaseVoucherRequest(BaseModel):
listing_id: str
quantity: int = Field(1, gt=0)

class PurchaseProposalResponse(BaseModel):
proposal: Dict[str, Any]
requires_approval: bool
message: str

class ApproveProposalRequest(BaseModel):
proposal: Dict[str, Any]
approved: bool
notes: Optional[str] = None

class RedeemVoucherRequest(BaseModel):
voucher_code: str
location: Optional[Dict[str, Any]] = None
notes: Optional[str] = None

class ValidationResponse(BaseModel):
valid: bool
reason: Optional[str] = None
voucher: Optional[Dict[str, Any]] = None
requires_approval: bool

class RedemptionResponse(BaseModel):
success: bool
message: str
voucher: Optional[Dict[str, Any]] = None

class SearchVouchersRequest(BaseModel):
voucher_type: Optional[VoucherTypeEnum] = None
category: Optional[str] = None
max_price_rays: Optional[int] = None
location_near: Optional[Dict[str, float]] = None

```
    return ValidationResponse(

Search for available voucher listings
“””

category=request.category,
location_near=request.location_near,
max_price_rays=request.max_price_rays
)

```
    return {
        "results": [
            {
                "id": listing.id,
                "title": listing.title,
                "description": listing.description,
                "type": listing.voucher_type.value,
                "price_rays": listing.price_rays,
                "energy_kwh": float(listing.energy_kwh) if listing.energy_kwh else None,
                "location": listing.redemption_location,
                "available": (listing.quantity_available - listing.quantity_sold) if listing.quantity_available else None,

            }
            for listing in listings
        ],
        "count": len(listings)
    }

        detail=str(e)
    )
