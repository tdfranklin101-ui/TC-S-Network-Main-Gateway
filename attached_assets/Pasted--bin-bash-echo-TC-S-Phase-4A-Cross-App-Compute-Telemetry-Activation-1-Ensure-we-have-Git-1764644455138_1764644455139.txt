#!/bin/bash
echo "🌞 TC-S Phase 4A: Cross-App Compute Telemetry Activation"

# 1. Ensure we have GitHub CLI auth
gh auth status || gh auth login

# --- SHARED TELEMETRY MODULE ADDITION ---
echo "📡 Adding telemetry emitter to TC-S-Network-Apps-Shared..."

gh repo clone tdfranklin101-ui/TC-S-Network-Apps-Shared shared-tmp
cd shared-tmp

mkdir -p lib

cat > lib/telemetry.js << 'EOF'
export async function sendTelemetry({ app, kWh, solar, endpoint }) {
  try {
    await fetch(endpoint, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        app,
        kWh,
        solar,
        ts: Date.now()
      })
    });
  } catch (err) {
    console.error("Telemetry error:", err);
  }
}
EOF

git add .
git commit -m "Add telemetry emitter (Phase 4A)"
git push origin main

cd ..
rm -rf shared-tmp

# --- UPDATE SOLAR DASHBOARD WITH UNIFIED TELEMETRY INGEST ROUTE ---
echo "🔧 Updating Solar Dashboard ingest endpoint..."

gh repo clone tdfranklin101-ui/TC-S-Network-Solar-Dashboard solar-tmp
cd solar-tmp

mkdir -p app/api/telemetry

cat > app/api/telemetry/route.ts << 'EOF'
import { NextResponse } from "next/server";

export async function POST(req: Request) {
  const data = await req.json();

  // Example structure:
  // { app: "Identify-Anything", kWh: 0.0021, solar: 0.00000043, ts: 1733000000000 }

  console.log("📥 Telemetry received:", data);

  // TODO: Insert into DB (Neon or Supabase)
  // TODO: Feed SI/CSI/ESI derivations

  return NextResponse.json({ ok: true });
}
EOF

git add .
git commit -m "Add telemetry ingest endpoint (Phase 4A)"
git push origin main

cd ..
rm -rf solar-tmp

# --- INJECT TELEMETRY INTO ALL 13 SATELLITE REPOS ---
echo "🌐 Updating all satellite apps to send compute telemetry..."

REPOS=(
"TC-S-Network-Identify-Anything"
"TC-S-Network-Market-Grid"
"TC-S-Network-Satellite-ID-Anywhere"
"TC-S-Network-Seismic-ID-Anywhere"
"TC-S-Network-Solar-Dashboard"
"TC-S-Network-Wallet"
"TC-S-Network-Solar-Reserve"
"TC-S-Network-GBI-Onboarding"
"TC-S-Network-Compute-Governance"
"TC-S-Network-Ethics-Engine"
"TC-S-Network-UIM-Protocol"
"TC-S-Network-Standards"
"TC-S-Network-Z-Private"
)

for repo in "${REPOS[@]}"
do
  echo "🔧 Updating $repo..."

  gh repo clone tdfranklin101-ui/$repo tmp-$repo
  cd tmp-$repo

  # Patch WPC usage file (auto-inject telemetry import + call)
  grep -Rl "computeAll" . | while read file; do
    # Add telemetry import
    sed -i '' '1s;^;import { sendTelemetry } from "TC-S-Network-Apps-Shared/lib/telemetry.js";\n;' "$file"

    # Inject telemetry send after computeAll()
    sed -i '' 's/computeAll(/const result = computeAll(/g' "$file"
    sed -i '' 's/);$/);\nsendTelemetry({ app: "'$repo'", kWh: result.kWh, solar: result.solar, endpoint: process.env.NEXT_PUBLIC_TELEMETRY_ENDPOINT });/g' "$file"
  done

  git add .
  git commit -m "Add compute telemetry emitter (Phase 4A)"
  git push origin main

  cd ..
  rm -rf tmp-$repo
done

echo "🎉 Phase 4A Complete — Telemetry Activated"
