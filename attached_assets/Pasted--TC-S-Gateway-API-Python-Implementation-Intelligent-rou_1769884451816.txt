# Request/Response Models

class UserType(str, Enum):
INDIVIDUAL = “individual”
ORGANIZATION = “organization”
EXPLORER = “explorer”

class Intent(str, Enum):
TRADE = “trade”
CREATE = “create”
COMMISSION = “commission”
LEARN = “learn”

class OrganizationType(str, Enum):
FESTIVAL = “festival”
CAMPUS = “campus”
COWORKING = “coworking”
COMMUNITY = “community”

class Timeline(str, Enum):
IMMEDIATE = “immediate”
PLANNING = “planning”
EXPLORING = “exploring”

class Budget(str, Enum):
PILOT = “pilot”
ANNUAL = “annual”
ENTERPRISE = “enterprise”

class Context(BaseModel):
organization_type: Optional[OrganizationType] = None
participant_count: Optional[int] = None
timeline: Timeline = Timeline.EXPLORING
budget: Optional[Budget] = None

class RouteRequest(BaseModel):
user_type: Optional[UserType] = None
intent: Optional[Intent] = None
context: Optional[Context] = None

class OnboardingStep(BaseModel):
id: str
title: str
description: str
action: str
estimated_time: int  # seconds

class RouteResponse(BaseModel):
path: str
route: str
network: Optional[str]
onboarding: Dict[str, Any]
recommendations: List[str]
assistant_message: str
pilot_details: Optional[Dict[str, Any]] = None

class IntentRequest(BaseModel):
message: str

class IntentResponse(BaseModel):
detected_intent: str
confidence: float
suggested_path: Optional[RouteResponse] = None
clarifying_questions: Optional[List[str]] = None

def extract_context(message: str) -> Context:
“””
Extracts context clues from natural language
“””
msg = message.lower()

```
org_type = None
if re.search(r'festival', msg):
    org_type = OrganizationType.FESTIVAL
elif re.search(r'campus|university|college', msg):
    org_type = OrganizationType.CAMPUS
elif re.search(r'coworking|workspace', msg):
    org_type = OrganizationType.COWORKING
elif re.search(r'farm|permaculture|regenerative', msg):
    org_type = OrganizationType.COMMUNITY

timeline = Timeline.EXPLORING
if re.search(r'asap|immediately|soon|next month', msg):
    timeline = Timeline.IMMEDIATE
elif re.search(r'planning|considering', msg):
    timeline = Timeline.PLANNING

budget = None
if re.search(r'pilot|test|proof', msg):
    budget = Budget.PILOT
elif re.search(r'annual|year|ongoing', msg):
    budget = Budget.ANNUAL
elif re.search(r'enterprise|large scale|custom', msg):
    budget = Budget.ENTERPRISE

# Extract participant count
participant_count = None
numbers = re.findall(r'\d+', message)
if numbers:
    participant_count = int(numbers[0])

return Context(
    organization_type=org_type,
    participant_count=participant_count,
    timeline=timeline,
    budget=budget
)
```
