async def create_listing(
    self,
    vendor_id: str,
    title: str,
    description: str,
    voucher_type: VoucherType,
    price_rays: int,
    redemption_location: str,
    redemption_instructions: str,
    **kwargs
) -> Dict[str, Any]:
    """
    Vendor creates a voucher listing
    
    Returns proposal for human approval
    """
    
    # Calculate energy footprint using LifeLens
    energy_estimate = await self._calculate_voucher_energy(
        voucher_type=voucher_type,
        item_description=description,
        location=redemption_location,
        **kwargs
    )
    
    # Ethics check
    ethics_result = await self._check_ethics({
        "action": "create_voucher_listing",
        "vendor_id": vendor_id,
        "voucher_type": voucher_type.value,
        "price_rays": price_rays,
        "description": description
    })
    
    # Prepare listing proposal
    proposal = {
        "listing_data": {
            "vendor_id": vendor_id,
            "title": title,
            "description": description,
            "voucher_type": voucher_type,
            "price_rays": price_rays,
            "energy_kwh": energy_estimate["total_kwh"],
            "redemption_location": redemption_location,
            "redemption_instructions": redemption_instructions,
            **kwargs
        },
        "energy_analysis": energy_estimate,
        "ethics_check": ethics_result,
        "suggested_price_rays": self._kwh_to_rays(energy_estimate["total_kwh"]),
        "requires_approval": ethics_result["score"] < 0.95
    }
    
    return proposal

async def purchase_voucher(
    self,
    listing_id: str,
    buyer_id: str,
    quantity: int = 1
) -> Dict[str, Any]:
    """
    Buyer purchases voucher
    Returns proposal for human approval
    """
    
    listing = self.db.query(VoucherListing).filter(
        VoucherListing.id == listing_id
    ).first()
    
    if not listing:
        raise ValueError("Listing not found")
    
    if not listing.active:
        raise ValueError("Listing is not active")
    
    # Check availability
    if listing.quantity_available is not None:
        remaining = listing.quantity_available - listing.quantity_sold
        if remaining < quantity:
            raise ValueError(f"Only {remaining} vouchers available")
    
    # Check buyer has sufficient funds
    total_cost = listing.price_rays * quantity
    buyer_balance = await self.wallet_service.get_balance(buyer_id)
    
    if buyer_balance < total_cost:
        raise ValueError(f"Insufficient funds. Need {total_cost} Rays, have {buyer_balance}")
    
    # Calculate expiration
    if listing.redemption_window_hours:
        expires_at = datetime.utcnow() + timedelta(hours=listing.redemption_window_hours)
    elif listing.valid_until:
        expires_at = listing.valid_until
    else:
        expires_at = datetime.utcnow() + timedelta(days=365)  # Default 1 year
    
    # Ethics check
    ethics_result = await self._check_ethics({
        "action": "purchase_voucher",
        "buyer_id": buyer_id,
        "listing_id": listing_id,
        "quantity": quantity,
        "total_cost_rays": total_cost
    })
    
    # Prepare purchase proposal
    proposal = {
        "listing": {
            "id": listing.id,
            "title": listing.title,
            "type": listing.voucher_type.value,
            "vendor": listing.vendor_id
        },
        "purchase": {
            "buyer_id": buyer_id,
            "quantity": quantity,
            "price_per_voucher": listing.price_rays,
            "total_cost": total_cost,
            "expires_at": expires_at.isoformat()
        },
        "ethics_check": ethics_result,
        "requires_approval": total_cost > 1000 or ethics_result["score"] < 0.90
    }
    
    return proposal

async def approve_purchase(self, proposal: Dict[str, Any]) -> List[Voucher]:
    """
    Human approves purchase, create vouchers and process payment
    """
    listing_id = proposal["listing"]["id"]
    buyer_id = proposal["purchase"]["buyer_id"]
    quantity = proposal["purchase"]["quantity"]
    total_cost = proposal["purchase"]["total_cost"]
    
    listing = self.db.query(VoucherListing).filter(
        VoucherListing.id == listing_id
    ).first()
    
    # Process payment via wallet service
    transaction = await self.wallet_service.transfer(
        from_user=buyer_id,
        to_user=listing.vendor_id,
        amount_rays=total_cost,
        description=f"Voucher purchase: {listing.title}"
    )
    
    # Create voucher instances
    vouchers = []
    for i in range(quantity):
        voucher_code = self._generate_voucher_code()
        qr_code = self._generate_qr_code(voucher_code)
        
        voucher = Voucher(
            voucher_code=voucher_code,
            listing_id=listing.id,
            buyer_id=buyer_id,
            vendor_id=listing.vendor_id,
            transaction_id=transaction.id,
            price_paid_rays=listing.price_rays,
            expires_at=datetime.fromisoformat(proposal["purchase"]["expires_at"]),
            qr_code_data=qr_code,
            status=VoucherStatus.ACTIVE
        )
        
        self.db.add(voucher)
        vouchers.append(voucher)
    
    # Update listing quantity
    listing.quantity_sold += quantity
    
    self.db.commit()
    
    for v in vouchers:
        self.db.refresh(v)
    
    # Notify vendor
    await self._notify_vendor_purchase(listing.vendor_id, vouchers)
    
    return vouchers

async def validate_redemption(
    self,
    voucher_code: str,
    vendor_id: str,
    location: Optional[Dict[str, Any]] = None
) -> Dict[str, Any]:
    """
    Vendor scans voucher, validate before redemption
    Returns validation result for human approval
    """
    
    voucher = self.db.query(Voucher).filter(
        Voucher.voucher_code == voucher_code
    ).first()
    
    if not voucher:
        return {
            "valid": False,
            "reason": "Voucher code not found",
            "voucher": None
        }
    
    # Check vendor authorization
    if voucher.vendor_id != vendor_id:
        return {
            "valid": False,
            "reason": "This voucher belongs to a different vendor",
            "voucher": voucher
        }
    
    # Check status
    if voucher.status != VoucherStatus.ACTIVE:
        return {
            "valid": False,
            "reason": f"Voucher is {voucher.status.value}",
            "voucher": voucher,
            "redeemed_at": voucher.redeemed_at if voucher.status == VoucherStatus.REDEEMED else None
        }
    
    # Check expiration
    now = datetime.utcnow()
    if voucher.expires_at < now:
        # Check if vendor allows grace period
        settings = self._get_vendor_settings(vendor_id)
        grace_hours = settings.allow_late_redemption_grace_hours or 0
        grace_expires = voucher.expires_at + timedelta(hours=grace_hours)
        
        if now > grace_expires:
            return {
                "valid": False,
                "reason": f"Voucher expired on {voucher.expires_at.isoformat()}",
                "voucher": voucher
            }
    
    # Check listing validity
    listing = voucher.listing
    if listing.valid_from and listing.valid_from > now:
        settings = self._get_vendor_settings(vendor_id)
        if not settings.allow_early_redemption:
            return {
                "valid": False,
                "reason": f"Voucher not valid until {listing.valid_from.isoformat()}",
                "voucher": voucher
            }
    
    # Valid!
    return {
        "valid": True,
        "voucher": voucher,
        "listing": listing,
        "buyer": voucher.buyer,
        "item": listing.title,
        "redemption_instructions": listing.redemption_instructions,
        "requires_approval": True  # Always require human approval to redeem
    }

async def approve_redemption(
    self,
    voucher_code: str,
    vendor_id: str,
    location: Optional[Dict[str, Any]] = None,
    notes: Optional[str] = None
) -> Dict[str, Any]:
    """
    Human approves redemption, mark voucher as redeemed
    """
    
    voucher = self.db.query(Voucher).filter(
        Voucher.voucher_code == voucher_code
    ).first()
    
    # Update voucher status
    voucher.status = VoucherStatus.REDEEMED
    voucher.redeemed_at = datetime.utcnow()
    voucher.redeemed_by = vendor_id
    voucher.redemption_notes = notes
    
    if location:
        voucher.redemption_location_actual = location.get("name")
    
    # Create redemption log
    redemption_log = VoucherRedemption(
        voucher_id=voucher.id,
        attempted_by=vendor_id,
        attempted_at=datetime.utcnow(),
        success=True,
        location_lat=location.get("lat") if location else None,
        location_lon=location.get("lon") if location else None,
        location_name=location.get("name") if location else None,
        redemption_method=RedemptionMethod.QR_CODE,
        notes=notes
    )
    
    self.db.add(redemption_log)
    self.db.commit()
    self.db.refresh(voucher)
    
    # Notify buyer
    await self._notify_buyer_redemption(voucher.buyer_id, voucher)
    
    return {
        "success": True,
        "voucher": voucher,
        "redeemed_at": voucher.redeemed_at,
        "message": f"Voucher redeemed successfully: {voucher.listing.title}"
    }

def search_listings(
    self,
    voucher_type: Optional[VoucherType] = None,
    category: Optional[str] = None,
    location_near: Optional[Dict[str, float]] = None,
    max_price_rays: Optional[int] = None,
    available_only: bool = True
) -> List[VoucherListing]:
    """
    Search for voucher listings
    """
    query = self.db.query(VoucherListing)
    
    if available_only:
        query = query.filter(VoucherListing.active == True)
    
    if voucher_type:
        query = query.filter(VoucherListing.voucher_type == voucher_type)
    
    if category:
        query = query.filter(VoucherListing.category == category)
    
    if max_price_rays:
        query = query.filter(VoucherListing.price_rays <= max_price_rays)
    
    # TODO: Location-based search (requires geocoding)
    
    return query.all()

def get_user_vouchers(
    self,
    user_id: str,
    status: Optional[VoucherStatus] = None,
    include_expired: bool = False
) -> List[Voucher]:
    """
    Get vouchers owned by a user
    """
    query = self.db.query(Voucher).filter(Voucher.buyer_id == user_id)
    
    if status:
        query = query.filter(Voucher.status == status)
    
    if not include_expired:
        query = query.filter(Voucher.expires_at > datetime.utcnow())
    
    return query.order_by(Voucher.expires_at).all()

def get_vendor_vouchers(
    self,
    vendor_id: str,
    status: Optional[VoucherStatus] = None
) -> List[Voucher]:
    """
    Get vouchers sold by a vendor
    """
    query = self.db.query(Voucher).filter(Voucher.vendor_id == vendor_id)
    
    if status:
        query = query.filter(Voucher.status == status)
    
    return query.order_by(Voucher.purchased_at.desc()).all()

def _generate_voucher_code(self) -> str:
    """
    Generate unique voucher code like TC-8F3D2A
    """
    while True:
        code = f"TC-{secrets.token_hex(3).upper()}"
        existing = self.db.query(Voucher).filter(
            Voucher.voucher_code == code
        ).first()
        if not existing:
            return code

def _generate_qr_code(self, voucher_code: str) -> str:
    """
    Generate QR code as base64 encoded image
    """
    qr = qrcode.QRCode(version=1, box_size=10, border=5)
    qr.add_data(voucher_code)
    qr.make(fit=True)
    
    img = qr.make_image(fill_color="black", back_color="white")
    
    # Convert to base64
    buffer = io.BytesIO()
    img.save(buffer, format='PNG')
    img_str = base64.b64encode(buffer.getvalue()).decode()
    
    return f"data:image/png;base64,{img_str}"

    production_kwh = base_energy.get(voucher_type, 1.0)
    coordination_kwh = 0.05  # Digital voucher overhead
    
    total_kwh = production_kwh + coordination_kwh
    
    return {
        "total_kwh": total_kwh,
        "breakdown": {
            "production": production_kwh,
            "coordination": coordination_kwh
        }
    }

async def _check_ethics(self, action_data: Dict[str, Any]) -> Dict[str, Any]:
    """
    Run ethics check through TC-S Ethics Engine
    """
    if not self.ethics_engine:
        return {"score": 1.0, "approved": True}
    
    result = await self.ethics_engine.evaluate(action_data)
    return result

def _kwh_to_rays(self, kwh: float) -> int:
    """
    Convert kWh to Rays (10,000 Rays = 1 Solar = 4,913 kWh)
    """
    rays_per_kwh = 10000 / 4913
    return int(kwh * rays_per_kwh)
