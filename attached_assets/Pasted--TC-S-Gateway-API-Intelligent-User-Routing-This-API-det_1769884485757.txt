/**

- POST /api/gateway/intent
- 
- AI-powered intent detection from natural language
- 
- Request body:
- {
- “message”: “I want to start earning Solar” | “We need marketplace for our festival” | etc.
- }
- 
- Response:
- {
- “detectedIntent”: “trade” | “commission” | “learn”,
- “confidence”: 0.0-1.0,
- “suggestedPath”: {…same as /route response},
- “clarifyingQuestions”: […] // if confidence < 0.7
- }
  */

router.post(’/intent’, async (req, res) => {
const { message } = req.body;

// This would integrate with Kid Solar or another LLM for intent detection
const intent = await detectIntent(message);

if (intent.confidence < 0.7) {
return res.json({
detectedIntent: intent.type,
confidence: intent.confidence,
clarifyingQuestions: [
‘Are you joining as an individual or representing an organization?’,
‘What are you hoping to accomplish?’,
‘Do you want to participate in the existing network or create your own?’
]
});
}

const route = determineRoute(intent.userType, intent.intent, intent.context);

res.json({
detectedIntent: intent.type,
confidence: intent.confidence,
suggestedPath: route
});
});

async function detectIntent(message) {
// Intent detection logic (would integrate with LLM)
// For now, simple keyword matching

const msg = message.toLowerCase();

// Organization signals
if (msg.match(/festival|event|campus|university|venue|community|our organization|we need/)) {
return {
type: ‘commission’,
userType: ‘organization’,
intent: ‘commission’,
confidence: 0.85,
context: extractContext(message)
};
}

// Individual/creator signals
if (msg.match(/i want|join|earn|trade|create|sell|my wallet|daily solar/)) {
return {
type: ‘trade’,
userType: ‘individual’,
intent: ‘trade’,
confidence: 0.9,
context: null
};
}

// Explorer signals
if (msg.match(/learn|understand|how does|what is|whitepaper|curious/)) {
return {
type: ‘learn’,
userType: ‘explorer’,
intent: ‘learn’,
confidence: 0.8,
context: null
};
}

return {
type: ‘unknown’,
userType: null,
intent: null,
confidence: 0.3,
context: null
};
}

function extractContext(message) {
const msg = message.toLowerCase();

return {
organizationType:
msg.match(/festival/) ? ‘festival’ :
msg.match(/campus|university|college/) ? ‘campus’ :
msg.match(/coworking|workspace/) ? ‘coworking’ :
msg.match(/farm|permaculture|regenerative/) ? ‘community’ :
null,
participantCount: extractNumber(message),
timeline:
msg.match(/asap|immediately|soon|next month/) ? ‘immediate’ :
msg.match(/planning|considering|exploring/) ? ‘planning’ :
‘exploring’,
budget:
msg.match(/pilot|test|proof/) ? ‘pilot’ :
msg.match(/annual|year|ongoing/) ? ‘annual’ :
msg.match(/enterprise|large scale|custom/) ? ‘enterprise’ :
null
};
}

function extractNumber(text) {
const match = text.match(/\d+/);
return match ? parseInt(match[0]) : null;
}
